daemon.idle_timeout
  How long a daemon runs without a request before it stops.
  type: duration, such as 10m (units s, m, h, d)
  default: 30m
  file: .codix/config.toml

edits.generated
  Generated or vendored files, which edits refuse to change without --allow generated: paths or gitignore-style patterns, relative to the root.
  type: comma-separated directories, relative to the project root
  default: none
  file: codix.toml

index.max_file_size
  Files larger than this are skipped.
  type: size, such as 2MB (units B, KB, MB, GB; 1 KB = 1024 B)
  default: 2MB
  file: codix.toml

index.scan_interval
  How often a running daemon rescans the whole project, catching changes the watcher missed.
  type: duration, such as 10m (units s, m, h, d)
  default: 10m
  file: .codix/config.toml

languages.overrides."<glob>"
  Language of the files matching each glob, overriding detection.
  type: language, one of: c, cpp, csharp, go, java, javascript, json, kotlin, markdown, php, python, ruby, rust, sql, text, toml, tsx, typescript, xml, yaml
  default: none
  file: codix.toml

previews.expiry
  How long a preview can be applied after it is planned.
  type: duration, such as 10m (units s, m, h, d)
  default: 1h
  file: .codix/config.toml

sources.roots
  Source directories, overriding those found from build files.
  type: comma-separated directories, relative to the project root
  default: none
  file: codix.toml
